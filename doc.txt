
# Instructions

// MRI
D0: AND = AC ^ DR
D1: ADD = AC + DR, E <- Cout
D2: LDA = AC <- M[AR]
D3: STA = M[AR] <- AC
D4: BUN = PC <- AR
D5: BSA = M[AR] <- PC, PC <- AR + 1
D6: ISZ = M[AR] <- M[AR] + 1 if M[AR] + 1 = 0 then PC <- PC + 1
D7: SUB: Subtract = AC - DR, E <- Cout 
D8: LIX: Load Index Register = IX <- DR
D9: LIA: Load Index Register From Accumulator = IX <- AC
D10: INX: Increase Index Register = IX <- IX + AR
D11: JZ  : Jump if Zero     = if (AC = 0)  then PC <- AR
D12: JNZ : Jump if Not Zero = if (AC != 0) then PC <- AR
D13: JP  : Jump if Positive = if (AC > 0)  then PC <- AR
D14: JN  : Jump if Negative = if (AC < 0)  then PC <- AR

// RRI
B0: CLA = AC <- 0
B1: INC = AC <- AC + 1
B2: HLT = S <- 0

//IOI
B0: ION = R <- 1
B1: INP = AC(0-7) <- INPR, FGI <- 0
B2: OUT = OUTR <- AC(0-7), FGO <- 0

// Register Architecture

[ 00 0000 000000000000 ] = 18 bits

Addressing Bits = 2
Opcode Bits = 4
Address Bits = 12

RAM = 4096 * 18

# Registers

AC = 12 bit
PC = 12 bit
AR = 12 bit
DR = 12 bit
TR = 12 bit
IX: Index Register = 12 bit
IR = 18 bit
INPR = 8 bit
OUTR = 8 bit

# Addressing Mode
I1: 00 = Direct   = NOP
I2: 01 = Indirect = AR <- M[AR]
I3: 10 = Index    = AR <- M[AR + IX]
11 = NOP

# ALU

S1 S0 Cin
0  0  0    D = A + B          ; Add
0  0  1    D = A + B + 1      ; Add with carry
0  1  0    D = A + B'         ; Subtract with borrow
0  1  1    D = A + B' + 1     ; Subtract
1  0  0    D = A              ; Transfer A
1  0  1    D = A + 1          ; Increment A
1  1  0    D = A - 1          ; Decrement A
1  1  1    D = A              ; Transfer A
